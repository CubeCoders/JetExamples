cmake_minimum_required(VERSION 3.16)
project(teapot_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(teapot preview.cpp)
target_include_directories(teapot PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(teapot PRIVATE Jet)
if(MSVC)
    target_compile_options(teapot PRIVATE /UNDEBUG)
else()
    target_compile_options(teapot PRIVATE -UNDEBUG)
endif()
add_test(NAME teapot COMMAND teapot)
