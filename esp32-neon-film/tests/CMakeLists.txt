cmake_minimum_required(VERSION 3.16)
project(film_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(film preview.cpp)
target_include_directories(film PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(film PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(film PRIVATE /UNDEBUG)
else()
 target_compile_options(film PRIVATE -UNDEBUG)
endif()
add_test(NAME film COMMAND film)

option(JET_REFERENCE_RASTER "Disable fast material spans for reference comparisons" OFF)
if(JET_REFERENCE_RASTER)
 target_compile_definitions(Jet PRIVATE JET_FAST_SIMPLE_SPANS=0 JET_FAST_OPAQUE_UNLIT_SPANS=0 JET_SKIP_UNLIT_NORMALS=0)
endif()
foreach(test offscreen_clipping unlit_spans texture_options water_reflection painter_buckets)
 add_executable(${test} ../../components/Jet/tests/${test}.cpp)
 target_link_libraries(${test} PRIVATE Jet)
 if(MSVC)
  target_compile_options(${test} PRIVATE /UNDEBUG)
 else()
  target_compile_options(${test} PRIVATE -UNDEBUG)
 endif()
 add_test(NAME ${test} COMMAND ${test})
endforeach()

add_executable(minimal_overlay minimal_overlay.cpp)
target_include_directories(minimal_overlay PRIVATE ../../components/esp32_jet)
target_link_libraries(minimal_overlay PRIVATE Jet)
if(MSVC)
 target_compile_options(minimal_overlay PRIVATE /UNDEBUG)
else()
 target_compile_options(minimal_overlay PRIVATE -UNDEBUG)
endif()
add_test(NAME minimal_overlay COMMAND minimal_overlay)

add_executable(film_video video.cpp)
target_include_directories(film_video PRIVATE ../main)
target_link_libraries(film_video PRIVATE Jet)
