cmake_minimum_required(VERSION 3.16)
project(film_quality LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC .)
add_executable(film_quality quality.cpp)
target_include_directories(film_quality PRIVATE ../../main)
target_link_libraries(film_quality PRIVATE Jet)

enable_testing()
add_executable(perspective_precision ../../../components/Jet/tests/perspective_precision.cpp)
target_link_libraries(perspective_precision PRIVATE Jet)
if(MSVC)
 target_compile_options(perspective_precision PRIVATE /UNDEBUG)
else()
 target_compile_options(perspective_precision PRIVATE -UNDEBUG)
endif()
add_test(NAME perspective_precision COMMAND perspective_precision)
