cmake_minimum_required(VERSION 3.16)
project(texture_lab_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(texture_lab preview.cpp)
target_include_directories(texture_lab PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(texture_lab PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(texture_lab PRIVATE /UNDEBUG)
else()
 target_compile_options(texture_lab PRIVATE -UNDEBUG)
endif()
add_test(NAME texture_lab COMMAND texture_lab)
