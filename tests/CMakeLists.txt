cmake_minimum_required(VERSION 3.16)
project(jet_examples_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_subdirectory(../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../esp32-template-cube/main/firmware)
add_executable(frame_schedule frame_schedule.cpp)
add_executable(cube cube.cpp)
target_include_directories(cube PRIVATE ../esp32-template-cube/main ../components/esp32_jet)
target_link_libraries(cube PRIVATE Jet)
add_executable(sprite_scanline ../components/Jet/tests/sprite_scanline.cpp)
target_link_libraries(sprite_scanline PRIVATE Jet)
add_executable(performance_overlay performance_overlay.cpp)
target_include_directories(performance_overlay PRIVATE ../components/esp32_jet)
target_link_libraries(performance_overlay PRIVATE Jet)
foreach(test frame_schedule cube sprite_scanline performance_overlay)
    # Assertions are part of these tests, including in a Release build.
    if(MSVC)
        target_compile_options(${test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${test} COMMAND ${test})
endforeach()
add_executable(water_reflection ../components/Jet/tests/water_reflection.cpp)
target_link_libraries(water_reflection PRIVATE Jet)
if(MSVC)
    target_compile_options(water_reflection PRIVATE /UNDEBUG)
else()
    target_compile_options(water_reflection PRIVATE -UNDEBUG)
endif()
add_test(NAME water_reflection COMMAND water_reflection)
