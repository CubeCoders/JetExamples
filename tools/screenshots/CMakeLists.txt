cmake_minimum_required(VERSION 3.16)
project(jet_example_screenshots LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(EXAMPLES_ROOT "../.." ABSOLUTE)
file(GLOB JET_SOURCES CONFIGURE_DEPENDS "${EXAMPLES_ROOT}/components/Jet/src/*.cpp")
function(capture name header ns)
    set(config "${EXAMPLES_ROOT}/${name}/main/firmware")
    add_library(jet_${name} STATIC ${JET_SOURCES})
    target_include_directories(jet_${name} PUBLIC "${config}" "${EXAMPLES_ROOT}/components/Jet/src")
    add_executable(${name} capture.cpp)
    target_include_directories(${name} PRIVATE "${EXAMPLES_ROOT}/${name}/main" "${EXAMPLES_ROOT}/components/esp32_jet")
    target_compile_definitions(${name} PRIVATE DEMO_HEADER="${header}" DEMO_NAMESPACE=${ns} ${ARGN})
    target_link_libraries(${name} PRIVATE jet_${name})
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
endfunction()
