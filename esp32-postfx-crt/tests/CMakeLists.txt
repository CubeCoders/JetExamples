cmake_minimum_required(VERSION 3.16)
project(crt_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(crt_preview preview.cpp)
target_include_directories(crt_preview PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(crt_preview PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(crt_preview PRIVATE /UNDEBUG)
else()
 target_compile_options(crt_preview PRIVATE -UNDEBUG)
endif()
add_test(NAME crt_scene COMMAND crt_preview)
foreach(layout IN ITEMS full half field half_field disabled)
 set(half 0)
 set(field 0)
 set(enabled 1)
 if(layout MATCHES "half")
  set(half 1)
 endif()
 if(layout MATCHES "field")
  set(field 1)
 endif()
 if(layout STREQUAL "disabled")
  set(enabled 0)
 endif()
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${layout}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${layout}/JetConfig.hpp" "#pragma once\n#define POSTFX_CRT ${enabled}\n#define HALF_WIDTH_BUFFERS ${half}\n#define FIELD_BUFFERS ${field}\n#define CRT_SCANLINE_INTENSITY 112\n#define POSTFX_BLOOM 0\n#define POSTFX_ANTIALIASING 0\n#define POSTFX_MOTION_BLUR 0\n#define POSTFX_CHROMATIC 0\n#define POSTFX_PIXELATE 0\n")
 add_executable(crt_${layout} ../../components/Jet/tests/crt_layout.cpp ../../components/Jet/src/PostFX.cpp)
 target_include_directories(crt_${layout} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${layout}" ../../components/Jet/src)
 if(MSVC)
  target_compile_options(crt_${layout} PRIVATE /UNDEBUG)
 else()
  target_compile_options(crt_${layout} PRIVATE -UNDEBUG)
 endif()
 add_test(NAME crt_${layout} COMMAND crt_${layout})
endforeach()
