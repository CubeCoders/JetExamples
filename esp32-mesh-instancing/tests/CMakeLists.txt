cmake_minimum_required(VERSION 3.16)
project(instancing_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(instancing preview.cpp)
target_include_directories(instancing PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(instancing PRIVATE Jet Threads::Threads)
add_test(NAME instancing_comparison COMMAND instancing)
add_executable(mesh_instances ../../components/Jet/tests/mesh_instances.cpp)
target_link_libraries(mesh_instances PRIVATE Jet)
add_test(NAME engine_instances COMMAND mesh_instances)
# A separate library keeps every translation unit on the same configuration.
get_target_property(JET_SOURCES Jet SOURCES)
add_library(JetFeatures STATIC ${JET_SOURCES})
target_include_directories(JetFeatures PUBLIC features ../../components/Jet/src)
add_executable(mesh_instances_features ../../components/Jet/tests/mesh_instances.cpp)
target_link_libraries(mesh_instances_features PRIVATE JetFeatures)
add_test(NAME engine_instances_features COMMAND mesh_instances_features)
foreach(target instancing mesh_instances mesh_instances_features)
 if(MSVC)
  target_compile_options(${target} PRIVATE /UNDEBUG)
 else()
  target_compile_options(${target} PRIVATE -UNDEBUG)
 endif()
endforeach()
