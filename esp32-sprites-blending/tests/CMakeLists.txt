cmake_minimum_required(VERSION 3.16)
project(courtyard_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(courtyard preview.cpp)
target_include_directories(courtyard PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(courtyard PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(courtyard PRIVATE /UNDEBUG)
else()
 target_compile_options(courtyard PRIVATE -UNDEBUG)
endif()
add_test(NAME courtyard COMMAND courtyard)
