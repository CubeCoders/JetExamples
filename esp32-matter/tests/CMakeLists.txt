cmake_minimum_required(VERSION 3.16)
project(matter_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
foreach(name preview verify)
 add_executable(matter_${name} ${name}.cpp)
 target_include_directories(matter_${name} PRIVATE ../main)
 target_link_libraries(matter_${name} PRIVATE Jet Threads::Threads)
 if(MSVC)
  target_compile_options(matter_${name} PRIVATE /UNDEBUG)
 else()
  target_compile_options(matter_${name} PRIVATE -UNDEBUG)
 endif()
endforeach()
add_test(NAME exhibition_timeline COMMAND matter_verify)
add_executable(matter_video video.cpp)
target_include_directories(matter_video PRIVATE ../main)
target_link_libraries(matter_video PRIVATE Jet)
