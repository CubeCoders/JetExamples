cmake_minimum_required(VERSION 3.16)
project(depth_teapot_preview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(depth_teapot preview.cpp)
target_include_directories(depth_teapot PRIVATE ../main ../../components/esp32_jet)
target_link_libraries(depth_teapot PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(depth_teapot PRIVATE /UNDEBUG)
else()
 target_compile_options(depth_teapot PRIVATE -UNDEBUG)
endif()
add_test(NAME depth_teapot COMMAND depth_teapot)
foreach(test runtime_depth depth_buffer depth_sort wide_depth water_reflection)
 add_executable(${test} ../../components/Jet/tests/${test}.cpp)
 target_link_libraries(${test} PRIVATE Jet)
 if(MSVC)
  target_compile_options(${test} PRIVATE /UNDEBUG)
 else()
  target_compile_options(${test} PRIVATE -UNDEBUG)
 endif()
 add_test(NAME ${test} COMMAND ${test})
endforeach()
